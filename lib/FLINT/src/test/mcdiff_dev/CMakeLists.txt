# gpu-port-mcdiff unit test: host co_DS_expr against device co_DS_expr_dev (README.md).
# cmake -S . -B <build> -DCMAKE_Fortran_COMPILER=nvfortran -DORION_SRC=<ORION tree> -DOSLO_MODDIR=<dir> -DOSLO_LIBS=<list> [-DTT_NS=7|OFF] [-DACC=OFF]
cmake_minimum_required(VERSION 3.13)
project(mcdiff_dev_test LANGUAGES Fortran)
set(ORION_SRC "" CACHE PATH "ORION source tree (the checkout MOSE_GPU builds with)")
set(TT_NS "7" CACHE STRING "MOSE_TT_NS of the compile-time build (production MOSE_CT=ON, MOSE_NSC=7); OFF = universal build")
set(GPU_OPTS "cc70,cc80,cuda12.6" CACHE STRING "-gpu= options")
set(OSLO_MODDIR "" CACHE PATH "Directory with oslo.mod + interface_definitions.mod from the MOSE_GPU build (OSlo is not rebuilt here)")
set(OSLO_LIBS "" CACHE STRING "Prebuilt libOSlo.a and its link dependencies, in the order of the MOSE_GPU link line")
option(ACC "Device build (OFF: the same sources compiled for the host only, for the Python check without a GPU)" ON)
get_filename_component(FLINT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
file(GLOB ORION_F ${ORION_SRC}/src/fortran/lib/*.f90)
file(GLOB FLINT_F ${FLINT_ROOT}/src/lib/*.f90 ${FLINT_ROOT}/src/lib/Lib_ChemMech/*.f90)
set(CMAKE_Fortran_MODULE_DIRECTORY ${CMAKE_BINARY_DIR}/modules)
# Flags of the MOSE_GPU production device build (flags.make of the MOSEL target, RELEASE, USE_OPENACC=ON)
if(ACC)
  set(ACCF "-acc=gpu -gpu=${GPU_OPTS} -Minfo=accel")
  set(CUDAF "-cuda")
endif()
set(CMAKE_Fortran_FLAGS "-cpp -Mbackslash -tp=x86-64-v3 ${ACCF} -fast -O3 -unroll -Minline -Mipa ${CUDAF}")
add_library(orion_t STATIC ${ORION_F})
add_library(flint_t STATIC ${FLINT_F})
target_link_libraries(flint_t PUBLIC orion_t)
target_include_directories(flint_t PUBLIC ${OSLO_MODDIR})
if(NOT TT_NS STREQUAL "OFF")
  math(EXPR NZ "${TT_NS}+1")
  target_compile_definitions(flint_t PUBLIC MOSE_CHEM_INLINE MOSE_CHEM_NS MOSE_CHEM_NZ=${NZ} MOSE_CT MOSE_TT_NS=${TT_NS})
endif()
set_source_files_properties(${FLINT_ROOT}/src/lib/Lib_Radau5_dev.f90 PROPERTIES COMPILE_OPTIONS -Mnoinline)
add_executable(test_co_DS_dev test_co_DS_dev.F90)
target_link_libraries(test_co_DS_dev flint_t ${OSLO_LIBS})
# Linked with the device runtime in both variants: the prebuilt OSlo objects come from the device build.
set_target_properties(test_co_DS_dev PROPERTIES LINK_FLAGS "-acc=gpu -gpu=${GPU_OPTS} -cuda -c++libs")
